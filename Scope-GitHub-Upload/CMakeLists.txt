cmake_minimum_required(VERSION 3.16)
project(DevScope LANGUAGES CXX RC)
add_executable(DevScope WIN32 main.cpp app.rc)
target_compile_features(DevScope PRIVATE cxx_std_17)
target_compile_definitions(DevScope PRIVATE UNICODE _UNICODE)
target_link_libraries(DevScope PRIVATE comctl32 shell32 advapi32 psapi gdi32 ole32 version wevtapi uuid)
if(MINGW)
 target_link_options(DevScope PRIVATE -municode -static-libgcc -static-libstdc++)
endif()
